// each counted or stated token at the chosen date: what backs it and where, and which chains it is on
function bsTokens(){
  const HL = headline(), t = iso(), backs = (L.backing || []).filter(b => b.available <= t), out = [];
  for (const [id, x] of Object.entries(HL.by)){
    const bt = backs.filter(b => b.token === id), last = bt.reduce((m, b) => b.as_of > m ? b.as_of : m, '');
    let parts = bt.filter(b => b.as_of === last && b.amount_usd > 0).map(b => { const tk = b.asset_type === 'token', m = b.component.match(/\(([A-Za-z]+)\)/);
      const hk = b.held_at || (/named by/.test(b.held_label) ? '~code' : /^Not named/.test(b.held_label) ? '~none' : '~' + b.held_label);
      return {key: tk ? 'tok:' + (b.counterpart || b.id) : b.asset_type, tk, cp: b.counterpart, label: tk ? (b.counterpart ? ents[b.counterpart].name : m ? m[1] : b.component.split(',')[0]) : TVAT[b.asset_type],
        hk, hl: b.held_at ? b.held_label : hk === '~code' ? 'Fund named by code only' : hk === '~none' ? 'Custodian or bank not named' : b.held_label, hent: b.held_at, v: b.amount_usd, row: b}; });
    if (!parts.length) parts = [{key: 'unread', label: TVAT.unread, hk: '~none', hl: 'Custodian or bank not named', v: x.amount, row: null}];
    const R = parts.reduce((s, p) => s + p.v, 0), O = x.amount;
    const per = V.obs.filter(o => o.subject === id && o.chain !== 'all'), tot = V.obs.filter(o => o.subject === id && o.chain === 'all').sort((a, b) => a.period_end < b.period_end ? -1 : 1).pop();
    let use = tot ? per.filter(o => o.period_end === tot.period_end && o.grade === tot.grade) : [], mixed = false;
    if (!use.length){ const lt = {}; per.forEach(o => { if (!lt[o.chain] || o.period_end >= lt[o.chain].period_end) lt[o.chain] = o; }); use = Object.values(lt); mixed = !!tot && use.length > 0; }
    let ch = [], rest = 0;
    for (const o of use){ if (ents[o.chain]) ch.push({c: o.chain, label: ents[o.chain].name, v: o.value_usd, flow: 'f-' + id + '-on-' + o.chain, o}); else rest += o.value_usd; }
    const sum = ch.reduce((s, c) => s + c.v, 0) + rest;
    if (sum > O && sum){ const k = O / sum; ch.forEach(c => c.v *= k); rest *= k; } else rest += O - sum;
    if (rest > O * 0.0005) ch.push({c: '~rest', label: 'Other or not split', v: rest});
    out.push({id, e: ents[id], x, parts, ch, R, O, mixed, size: Math.max(R, O), ok: x.ok});
  }
  return out.sort((a, b) => b.size - a.size);
}
function bsPanel(toks, k, y0, X, bw){
  const N = [[], [], [], []], links = [], ix = {};
  toks.forEach((t, i) => ix[t.id] = i);
  const grp = (col, key, mk) => { let n = N[col].find(q => q.key === key); if (!n){ n = {col, key, v: 0, w: 0, ...mk}; N[col].push(n); } return n; };
  toks.forEach((t, i) => {
    const tn = {col: 2, key: t.id, tok: t, v: t.size, vL: t.R, vR: t.O, pos: i}; N[2].push(tn);
    for (const p of t.parts){ const b = grp(1, p.key, {label: p.label, tk: p.tk, cp: p.cp, unread: p.key === 'unread'}), h = grp(0, p.hk, {label: p.hl, ent: p.hent, none: p.hk === '~none', code: p.hk === '~code'});
      b.v += p.v; b.w += p.v * i; h.v += p.v; h.w += p.v * i; links.push({s: b, t: tn, v: p.v, kind: 'back', tok: t, p}); let hl = links.find(l => l.s === h && l.t === b); if (!hl){ hl = {s: h, t: b, v: 0, kind: 'held', toks: new Set()}; links.push(hl); } hl.v += p.v; hl.toks.add(t.id); }
    for (const c of t.ch){ const n = grp(3, c.c, {label: c.label, ent: c.c[0] === '~' ? null : c.c, rest: c.c === '~rest'}); n.v += c.v; n.w += c.v * i; links.push({s: tn, t: n, v: c.v, kind: 'chain', tok: t, c}); }
  });
  for (const col of [0, 1, 3]) N[col].forEach(n => n.pos = n.v ? n.w / n.v + (n.none || n.rest || n.unread ? 0.4 : 0) : 0);
  let H = 0;
  N.forEach((ns, col) => { ns.sort((a, b) => a.pos - b.pos); let y = y0; for (const n of ns){ n.x = X[col]; n.y = y; n.h = Math.max(1.5, n.v * k); y += Math.max(n.h, col === 2 ? 30 : 15) + (col === 2 ? 12 : 7); } H = Math.max(H, y - y0); });
  for (const n of N.flat()){ n.oo = 0; n.oi = 0; }
  const order = l => l.s.y + l.t.y / 1e4;
  for (const n of N.flat()){
    links.filter(l => l.s === n).sort((a, b) => a.t.y - b.t.y).forEach(l => { l.y0 = n.y + n.oo; n.oo += l.v * k; });
    links.filter(l => l.t === n).sort((a, b) => a.s.y - b.s.y).forEach(l => { l.y1 = n.y + n.oi; n.oi += l.v * k; });
  }
  return {N, links, H, bw, k};
}
function bsHot(l){
  if (!sel) return true;
  if (sel.kind === 'flow') return l.kind === 'chain' && l.c.flow === sel.id;
  const id = sel.id, e = ents[id];
  if (l.kind === 'held') return l.toks.has(id) || l.s.ent === id || (l.t.cp === id) || (e && e.kind !== 'product' && [...l.toks].some(t => ents[t].parent === id));
  return l.tok.id === id || ents[l.tok.id].parent === id || (l.kind === 'chain' && l.c.c === id) || (l.kind === 'back' && (l.p.cp === id || l.p.hent === id));
}

// who holds which roles: the contest is over collecting them
function bsRoleRows(){
  const F = V.flows, nm = id => ents[id].name, rows = [];
  for (const e of L.entities){ if (e.kind !== 'company' || !V.live.has(e.id)) continue; const id = e.id, kid = x => ents[x] && ents[x].parent === id;
    const r = [
      F.filter(f => f.type === 'issues' && f.payer === id).map(f => [nm(f.payee), f.id]),
      F.filter(f => (f.type === 'distributes' || ((f.type === 'partner' || f.type === 'owns') && ents[f.payee].kind === 'product')) && f.payer === id).map(f => [nm(f.payee), f.id]),
      F.filter(f => f.type === 'tokenizes' && f.payer === id).map(f => [nm(f.payee), f.id]),
      L.entities.filter(x => x.kind === 'ledger' && x.parent === id && V.live.has(x.id)).map(x => [x.name, null, x.id]),
      F.filter(f => ((f.type === 'custody' || f.type === 'manages') && (f.payee === id || kid(f.payee))) || (f.type === 'holding' && kid(f.payee) && ents[f.payee].kind === 'instrument')).map(f => [nm(f.payer) + (f.type === 'custody' ? ' custody' : ' reserves'), f.id]),
      F.filter(f => f.type === 'licensed' && f.payee === id).map(f => [nm(f.payer) + (f.status === 'announced' ? (/appl/i.test(f.label) ? ' (applied)' : ' (conditional)') : ''), f.id]),
      F.filter(f => (f.type === 'equity' && (f.payer === id || f.payee === id)) || (f.type === 'partner' && ents[f.payee].kind !== 'product' && (f.payer === id || f.payee === id))).map(f => [f.type === 'partner' ? 'with ' + nm(f.payer === id ? f.payee : f.payer) : (f.payer === id ? 'buys ' + nm(f.payee) : 'bought by ' + nm(f.payer)), f.id]),
    ];
    const n = r.slice(0, 6).filter(c => c.length).length; if (n || r[6].length) rows.push({e, r, n});
  }
  return rows.sort((a, b) => b.n - a.n || b.r.flat().length - a.r.flat().length || (a.e.name < b.e.name ? -1 : 1));
}

  const Wc = Math.max(820, TVEL.clientWidth - 6), Hc = Math.max(400, Math.min(580, TVEL.clientHeight - 16)), m = {l: 58, r: 96, t: 150, b: 30}, t = iso();
