#!/usr/bin/env python3
"""Buildout Ledger harness: one file builds every ledger in the series.

The AI profile below is the original AI Buildout Ledger harness, unchanged; the tokenization profile follows it.
Which one runs is decided by "profile" in raw/declared.json (absent means AI).

AI profile:

Stages:  collect (raw/xbrl_facts.csv + raw/declared.json, gathered by the agent)
         -> extract to schema -> verify (validate) -> diff -> write ledger.json

Usage:
  harness.py build  --asof YYYY-MM-DD --out ledger.json [--run-id ID]
  harness.py validate ledger.json
  harness.py backtest --from YYYY-MM-DD --to YYYY-MM-DD

Rules enforced:
  * Nothing reaches the ledger unless it is a row with source_url, grade, date.
  * A row is visible as of the day its source became public (filing date or
    announcement date), which is what --asof filters on.
  * Quarter values that a filing gives only as year-to-date are derived by
    subtraction and marked derived, with the arithmetic kept.
  * Where a filing gives both a quarter and a year-to-date figure, they must
    reconcile within TOL.
"""
import argparse, csv, json, sys, datetime as dt, re, os

HERE = os.path.dirname(os.path.abspath(__file__))
RAW = os.path.join(HERE, "..", "raw")
TOL = 2_000_000  # filings round to $1M; allow two units of rounding
GRADES = ["filed", "announced", "reported", "estimate"]
FLOW_TYPES = ["equity", "debt", "purchase_commitment", "capex", "product", "uses", "owns"]
KINDS = ["company", "product", "sink", "instrument", "investor", "government"]
NON_MONEY = ("product", "uses", "owns")
ACCESS = ["general", "limited", "preview"]
UNITS = ["usd", "users", "tokens"]
STATUSES = ["cash_moved", "contracted", "announced", "reported"]
CIK = {}  # filled from declared entities at load time


def d(s):
    return dt.date.fromisoformat(s)


def xbrl_url(entity, concept):
    return f"https://data.sec.gov/api/xbrl/companyconcept/CIK{CIK[entity]}/us-gaap/{concept}.json"


def load_raw():
    with open(os.path.join(RAW, "xbrl_facts.csv")) as f:
        facts = list(csv.DictReader(f))
    for r in facts:
        r["val"] = int(r["val"])
    with open(os.path.join(RAW, "declared.json")) as f:
        declared = json.load(f)
    for e in declared["entities"]:
        if e.get("cik"):
            CIK[e["id"]] = e["cik"]
    return facts, declared


def derive_observations(facts, concepts, asof):
    """Turn raw XBRL facts into per-period observations plus reconciliation checks."""
    obs, checks = [], []
    groups = {}
    for r in facts:
        if d(r["filed"]) > asof:
            continue
        groups.setdefault((r["entity"], r["concept"]), []).append(r)
    for (entity, concept), rows in sorted(groups.items()):
        meta = concepts[f"{entity}|{concept}"]
        src = xbrl_url(entity, concept)

        def mk(start, end, val, row, derived=None):
            return {
                "id": f"o-{entity}-{meta['metric']}-{end}",
                "subject": entity, "metric": meta["metric"], "concept": concept,
                "period_start": start, "period_end": end, "value_usd": val,
                "derived": bool(derived), "derivation": derived,
                "grade": "filed", "form": row["form"], "accn": row["accn"],
                "available": row["filed"], "as_of": row["filed"],
                "definition": meta["definition"], "source_url": src,
                "source_title": f"SEC XBRL, {row['form']} {row['accn']}",
            }

        instants = [r for r in rows if not r["start"]]
        for r in instants:
            obs.append(mk(None, r["end"], r["val"], r))
        durs = [r for r in rows if r["start"]]
        direct = {}
        for r in durs:
            days = (d(r["end"]) - d(r["start"])).days
            if 75 <= days <= 100:
                direct[r["end"]] = r
        by_start = {}
        for r in durs:
            by_start.setdefault(r["start"], []).append(r)
        quarters = {}
        for end, r in direct.items():
            quarters[end] = mk(r["start"], end, r["val"], r)
        for start, cum in by_start.items():
            cum.sort(key=lambda r: r["end"])
            prev = None
            for r in cum:
                days = (d(r["end"]) - d(start)).days
                if days > 380:  # trailing-twelve-month style fact, handled below
                    continue
                if prev is None:
                    prev = r
                    continue
                gap = (d(r["end"]) - d(prev["end"])).days
                if 75 <= gap <= 100:
                    q = r["val"] - prev["val"]
                    qstart = (d(prev["end"]) + dt.timedelta(days=1)).isoformat()
                    how = f"{r['val']} (to {r['end']}, {r['accn']}) minus {prev['val']} (to {prev['end']}, {prev['accn']})"
                    if r["end"] in direct:
                        diff = abs(direct[r["end"]]["val"] - q)
                        checks.append({
                            "check": "quarter_vs_ytd", "subject": entity, "metric": meta["metric"],
                            "period_end": r["end"], "reported_quarter": direct[r["end"]]["val"],
                            "ytd_difference": q, "abs_diff": diff, "pass": diff <= TOL,
                        })
                    else:
                        quarters[r["end"]] = mk(qstart, r["end"], q, r, derived=how)
                prev = r
        obs.extend(quarters[k] for k in sorted(quarters))
        # trailing-twelve-month facts: check against the sum of the four quarters we hold
        for r in durs:
            days = (d(r["end"]) - d(r["start"])).days
            starts_fy = any(x["start"] == r["start"] and x is not r for x in durs)
            if 350 <= days <= 380 and not starts_fy:
                ends = [e for e in sorted(quarters) if d(r["start"]) <= d(quarters[e]["period_start"]) and d(e) <= d(r["end"])]
                if len(ends) == 4:
                    s = sum(quarters[e]["value_usd"] for e in ends)
                    checks.append({
                        "check": "four_quarters_vs_ttm", "subject": entity, "metric": meta["metric"],
                        "period_end": r["end"], "reported_ttm": r["val"], "sum_of_quarters": s,
                        "abs_diff": abs(s - r["val"]), "pass": abs(s - r["val"]) <= TOL,
                    })
    return obs, checks


def ttm(obs, entity, metric):
    qs = sorted([o for o in obs if o["subject"] == entity and o["metric"] == metric and o["period_start"]],
                key=lambda o: o["period_end"])
    last = qs[-4:]
    return (sum(o["value_usd"] for o in last) if last else None, last)


def fmt(v):
    return f"${v/1e9:,.1f}B"


def names_of(dec):
    return {e["id"]: e["name"] for e in dec["entities"]}


def build(asof, run_id, with_history=True):
    facts, dec = load_raw()
    obs, checks = derive_observations(facts, dec["concepts"], asof)
    flows, events = [], []
    for f in dec["flows"]:
        if d(f["date"]) <= asof:
            g = {k: v for k, v in f.items() if k != "revisions"}
            g["available"] = f["date"]
            g["as_of"] = f["date"]
            # a revision is a later primary disclosure of the same deal; the newest one public by asof wins
            for r in sorted(f.get("revisions", []), key=lambda r: r["date"]):
                if d(r["date"]) > asof:
                    break
                g.update({k: r[k] for k in ("amount_usd", "amount_qualifier", "grade", "source_title", "source_url") if k in r})
                if r.get("status"): g["status"] = r["status"]
                g["available"] = g["as_of"] = r["date"]
                g["definition"] = f["definition"] + " Update " + r["date"] + ": " + r["note"]
                g["revised_from"] = f["source_url"]
                events.append({
                    "id": f"e-{f['id']}-r{r['date']}", "date": r["date"], "entities": [f["payer"], f["payee"]],
                    "flow": f["id"], "level": f["level"], "title": r["title"], "detail": r["note"],
                    "grade": r["grade"], "source_title": r["source_title"], "source_url": r["source_url"],
                })
            flows.append(g)
            if f["type"] in NON_MONEY:
                continue
            events.append({
                "id": "e-" + f["id"], "date": f["date"], "entities": [f["payer"], f["payee"]],
                "flow": f["id"], "level": f["level"], "title": f["title"], "detail": f["definition"],
                "grade": f["grade"], "source_title": f["source_title"], "source_url": f["source_url"],
            })
    # model releases: events on the product (or the lab, for labs without a separate app on the map)
    releases = [dict(r) for r in dec.get("releases", []) if d(r["date"]) <= asof]
    for r in releases:
        price = f" at ${r['price_in']:g} / ${r['price_out']:g} per 1M tokens" if r.get("price_in") is not None else ""
        events.append({
            "id": "e-" + r["id"], "date": r["date"], "entities": sorted({r["entity"], r["lab"]}), "flow": None, "release": r["id"],
            "level": "map" if r["material"] else "entity",
            "title": f"{r['model']} released{' (limited)' if r['access'] == 'limited' else ' (preview)' if r['access'] == 'preview' else ''}{', open weights' if r['open_weights'] else ''}{price}",
            "detail": r["note"], "grade": r["grade"], "source_title": r["source_title"], "source_url": r["source_url"],
        })
    # stated metrics: revenue, run-rate, users. Company-stated figures are 'announced'; filings are 'filed'
    mdef = dec.get("metric_defaults", {})
    metrics = []
    for m in dec.get("metrics", []):
        if d(m["available"]) > asof:
            continue
        x = {**{k: v for k, v in mdef.get(m["metric"], {}).items()}, **m}
        metrics.append(x)
    mnames = {"revenue_quarter": "revenue", "run_rate": "run-rate revenue", "coding_run_rate": "Claude Code run-rate", "weekly_users": "weekly users",
              "monthly_users": "monthly users", "subscribers": "subscribers", "api_tokens_per_min": "API tokens per minute", "purchase_commitments": "purchase commitments", "backlog": "contracted backlog"}
    def mval(m):
        v = m["value"]
        if m["unit"] == "usd": return fmt(v)
        return f"{v/1e9:,.2f}B".replace(".00B", "B") if v >= 1e9 else f"{v/1e6:,.0f}M"
    for m in metrics:
        q = {"more than": "more than ", "about": "about ", "under": "under ", "nearly": "nearly ", "at least": "at least "}.get(m.get("qualifier") or "", "")
        events.append({
            "id": "e-" + m["id"], "date": m["available"], "entities": [m["entity"]], "flow": None, "metric": m["id"], "level": "entity",
            "title": f"{names_of(dec)[m['entity']]} {mnames.get(m['metric'], m['metric'])}: {q}{mval(m)}" + (f" for quarter ending {m['as_of']}" if m["metric"] == "revenue_quarter" else f" as of {m['as_of']}"),
            "detail": m.get("definition", ""), "grade": m["grade"], "source_title": m["source_title"], "source_url": m["source_url"],
        })
    mm = {m["id"]: m for m in metrics}
    for c in dec.get("metric_checks", []):
        need_ids = [c.get("a"), c.get("b")] + c.get("parts", [])
        if not all(i in mm for i in need_ids if i):
            continue
        if c["test"] == "yoy_pct":
            got = (mm[c["a"]]["value"] / mm[c["b"]]["value"] - 1) * 100
            ok = abs(got - c["stated"]) <= c["tolerance"]
        elif c["test"] == "rounds_to":
            got = mm[c["a"]]["value"]; ok = round(got / c["round_to"]) * c["round_to"] == c["stated"]
        else:
            got = sum(mm[i]["value"] for i in c["parts"]); ok = round(got / c["round_to"]) * c["round_to"] == c["stated"]
        checks.append({"check": c["test"], "id": c["id"], "subject": c["entity"], "metric": "revenue_quarter", "period_end": max(mm[i]["as_of"] for i in need_ids if i),
                       "computed": got, "stated": c["stated"], "abs_diff": abs(got - c["stated"]), "pass": ok, "basis": c["basis"], "source_url": c["source_url"]})
    # periodic cash flows built from filed observations
    names = {e["id"]: e["name"] for e in dec["entities"]}
    for ent in [e["id"] for e in dec["entities"]]:
        total, last = ttm(obs, ent, "capex")
        if not last:
            continue
        latest = last[-1]
        flows.append({
            "id": f"f-{ent}-capex", "payer": ent, "payee": "plant", "type": "capex", "status": "cash_moved",
            "amount_usd": total,
            "amount_qualifier": "trailing four quarters" if len(last) == 4 else f"{len(last)} quarter(s) on record",
            "date": latest["available"], "available": latest["available"], "as_of": latest["period_end"],
            "grade": "filed", "definition": latest["definition"], "source_title": latest["source_title"],
            "source_url": latest["source_url"], "level": "map",
            "derived": True,
            "derivation": "sum of observations " + ", ".join(o["id"] for o in last),
            "title": f"{names[ent]} capital spending", "label": "Capex, last 4 qtrs" if len(last) == 4 else "Capex to date",
        })
    debt = [r for r in facts if r["entity"] == "oracle" and r["concept"] == "ProceedsFromIssuanceOfSeniorLongTermDebt" and d(r["filed"]) <= asof]
    if debt:
        latest = {"definition": dec["concepts"]["oracle|ProceedsFromIssuanceOfSeniorLongTermDebt"]["definition"],
                  "source_url": xbrl_url("oracle", "ProceedsFromIssuanceOfSeniorLongTermDebt")}
        raw_cum = max((r for r in facts if r["entity"] == "oracle" and r["concept"] == "ProceedsFromIssuanceOfSeniorLongTermDebt" and d(r["filed"]) <= asof),
                      key=lambda r: r["end"])
        flows.append({
            "id": "f-orcl-debt", "payer": "oracle-notes", "payee": "oracle", "type": "debt", "status": "cash_moved",
            "amount_usd": raw_cum["val"], "amount_qualifier": f"fiscal year to {raw_cum['end']}",
            "date": raw_cum["filed"], "available": raw_cum["filed"], "as_of": raw_cum["end"],
            "grade": "filed", "definition": latest["definition"],
            "source_title": f"SEC XBRL, {raw_cum['form']} {raw_cum['accn']}", "source_url": latest["source_url"],
            "level": "map", "title": "Oracle borrows through senior notes", "label": "Senior notes",
        })
    for e in dec["events"]:
        if d(e["date"]) <= asof:
            events.append(dict(e))
    # one event per filed observation: this is the entity-level "everything" stream
    by_series = {}
    for o in obs:
        by_series.setdefault((o["subject"], o["metric"]), []).append(o)
    label = {"capex": "capital spending", "revenue": "revenue", "rpo": "contracted backlog", "debt_proceeds": "senior debt raised"}
    for (ent, metric), series in by_series.items():
        series.sort(key=lambda o: o["period_end"])
        for i, o in enumerate(series):
            prev = series[i - 1] if i else None
            chg = ""
            if prev and prev["value_usd"]:
                pct = (o["value_usd"] - prev["value_usd"]) / prev["value_usd"] * 100
                chg = f" ({pct:+.0f}% vs prior period)"
            structural = False
            if metric == "rpo" and prev and o["value_usd"] - prev["value_usd"] > 100e9:
                structural = True
            events.append({
                "id": "e-" + o["id"], "date": o["available"], "entities": [ent], "flow": None,
                "observation": o["id"], "level": "map" if structural else "entity",
                "title": f"{names[ent]} {label[metric]} {fmt(o['value_usd'])} for period ending {o['period_end']}{chg}",
                "detail": o["definition"] + (" Derived: " + o["derivation"] if o["derived"] else ""),
                "grade": "filed", "source_title": o["source_title"], "source_url": o["source_url"],
            })
    events.sort(key=lambda e: (e["date"], e["id"]))
    ranges = [r for r in dec["ranges"] if d(r["period_end"]) <= asof and any(o["subject"] == r["subject"] and o["period_end"] == r["period_end"] for o in obs)]
    ents = [e for e in dec["entities"] if any(e["id"] in (f["payer"], f["payee"]) for f in flows) or any(o["subject"] == e["id"] for o in obs) or any(e["id"] in ev["entities"] for ev in events)]
    live = {e["id"] for e in ents}
    for e in list(ents):   # a product on the map brings its maker
        if e.get("parent") and e["parent"] not in live:
            ents.append(next(x for x in dec["entities"] if x["id"] == e["parent"])); live.add(e["parent"])
    # cross-check: edges that other public maps draw, and whether this ledger has a row for each
    coverage = []
    for r in dec.get("reference_edges", []):
        rows = [f["id"] for f in flows if {f["payer"], f["payee"]} == {r["a"], r["b"]}]
        missing_player = [x for x in (r["a"], r["b"]) if x not in {e["id"] for e in dec["entities"]}]
        # an edge another map draws can be left off on purpose (talks only, outside the window); it says why
        coverage.append({**r, "caught": bool(rows), "rows": rows, "excluded": bool(r.get("not_drawn")) and not rows,
                         "reason": "" if rows else (r["not_drawn"] if r.get("not_drawn") else "player not on the map: " + ", ".join(missing_player) if missing_player else "no sourced row yet")})
    ledger = {
        "schema_version": 1,
        "as_of": asof.isoformat(),
        "schema": {
            "grades": {"filed": "Regulatory filing (SEC XBRL)", "announced": "Company's own announcement", "reported": "Press report; no primary source", "estimate": "Third-party estimate"},
            "flow_types": FLOW_TYPES, "statuses": STATUSES,
            "levels": {"map": "Structural shift: changes who can do what or how capital moves", "entity": "Any recorded change at one entity"},
        },
        "entities": ents, "flows": flows, "observations": obs, "events": events, "ranges": ranges,
        "releases": releases, "metrics": metrics, "coverage": coverage,
        "stress": dec.get("stress", {}),
        "checks": checks, "corrections": dec.get("corrections", []),
        "runs": [{"id": run_id, "as_of": asof.isoformat(), "rows": len(flows) + len(obs) + len(events) + len(releases) + len(metrics)}],
        "watchlist": [{"entity": k.split("|")[0], "concept": k.split("|")[1], "url": xbrl_url(*k.split("|"))} for k in dec["concepts"]],
        "gaps": dec.get("gaps", []),
    }
    if with_history:
        # amount history per flow, so a reader can scrub time without re-deriving anything
        declared = {f["date"] for f in dec["flows"]} | {r["date"] for f in dec["flows"] for r in f.get("revisions", [])}
        dates = sorted({r["filed"] for r in facts if d(r["filed"]) <= asof} | {f["available"] for f in flows} | {x for x in declared if d(x) <= asof})
        hist = {f["id"]: [] for f in flows}
        for ds in dates:
            snap = build(d(ds), run_id, with_history=False)
            for f in snap["flows"]:
                h = hist.get(f["id"])
                if h is None:
                    continue
                if not h or h[-1]["amount_usd"] != f["amount_usd"] or h[-1]["as_of"] != f["as_of"]:
                    h.append({"available": ds, "amount_usd": f["amount_usd"], "as_of": f["as_of"],
                              "amount_qualifier": f["amount_qualifier"], "source_title": f["source_title"]})
        for f in flows:
            f["history"] = hist[f["id"]]
    return ledger


def validate(ledger):
    errs = []
    ids = set()
    ent_ids = {e["id"] for e in ledger["entities"]}

    def need(row, kind, fields):
        for f in fields:
            if row.get(f) in (None, ""):
                errs.append(f"{kind} {row.get('id')}: missing {f}")
        if row.get("grade") not in GRADES:
            errs.append(f"{kind} {row.get('id')}: bad grade {row.get('grade')}")
        u = row.get("source_url") or ""
        if not re.match(r"^https://[^\s]+\.[^\s]+$", u):
            errs.append(f"{kind} {row.get('id')}: bad source_url")
        if row.get("id") in ids:
            errs.append(f"duplicate id {row.get('id')}")
        ids.add(row.get("id"))

    for e in ledger["entities"]:
        if e.get("kind") not in KINDS: errs.append(f"entity {e['id']}: bad kind {e.get('kind')}")
        if e.get("parent") and e["parent"] not in ent_ids: errs.append(f"entity {e['id']}: unknown parent")
    for f in ledger["flows"]:
        if f.get("type") in NON_MONEY and f.get("amount_usd") is not None: errs.append(f"flow {f['id']}: a {f['type']} link cannot carry an amount")
        need(f, "flow", ["id", "payer", "payee", "type", "status", "date", "as_of", "definition", "source_url", "source_title"])
        if f.get("type") not in FLOW_TYPES: errs.append(f"flow {f['id']}: bad type")
        if f.get("status") not in STATUSES: errs.append(f"flow {f['id']}: bad status")
        if f.get("payer") not in ent_ids or f.get("payee") not in ent_ids: errs.append(f"flow {f['id']}: unknown entity")
        if f.get("amount_usd") is None and not f.get("amount_qualifier"): errs.append(f"flow {f['id']}: no amount and no qualifier")
        if f.get("grade") == "filed" and "sec.gov" not in f.get("source_url", ""): errs.append(f"flow {f['id']}: graded filed without an SEC source")
    for o in ledger["observations"]:
        need(o, "observation", ["id", "subject", "metric", "period_end", "value_usd", "accn", "available", "definition", "source_url"])
        if o["derived"] and not o.get("derivation"): errs.append(f"observation {o['id']}: derived without arithmetic")
    for e in ledger["events"]:
        need(e, "event", ["id", "date", "entities", "level", "title", "source_url"])
        for x in e.get("entities", []):
            if x not in ent_ids: errs.append(f"event {e['id']}: unknown entity {x}")
    for r in ledger.get("releases", []):
        need(r, "release", ["id", "entity", "lab", "model", "date", "access", "note", "source_url", "source_title"])
        if r["entity"] not in ent_ids or r["lab"] not in ent_ids: errs.append(f"release {r['id']}: unknown entity")
        if r["access"] not in ACCESS: errs.append(f"release {r['id']}: bad access")
        if (r.get("price_in") is None) != (r.get("price_out") is None): errs.append(f"release {r['id']}: price needs both input and output")
        if not isinstance(r.get("open_weights"), bool) or not isinstance(r.get("material"), bool): errs.append(f"release {r['id']}: open_weights and material must be true or false")
        if r.get("price_source_url") and not r["price_source_url"].startswith("https://"): errs.append(f"release {r['id']}: bad price_source_url")
    for m in ledger.get("metrics", []):
        need(m, "metric", ["id", "entity", "metric", "value", "unit", "as_of", "available", "definition", "source_url", "source_title"])
        if m["entity"] not in ent_ids: errs.append(f"metric {m['id']}: unknown entity")
        if m["unit"] not in UNITS: errs.append(f"metric {m['id']}: bad unit")
        if m["as_of"] > m["available"]: errs.append(f"metric {m['id']}: describes a date after it was published")
        if m["grade"] == "filed" and "sec.gov" not in m["source_url"]: errs.append(f"metric {m['id']}: graded filed without an SEC source")
    for c in ledger["checks"]:
        if not c["pass"]:
            errs.append(f"reconciliation failed: {c}")
    # derived flows must recompute from the observations they cite
    omap = {o["id"]: o for o in ledger["observations"]}
    for f in ledger["flows"]:
        if f.get("derivation") and f["derivation"].startswith("sum of observations "):
            parts = f["derivation"][len("sum of observations "):].split(", ")
            s = sum(omap[p]["value_usd"] for p in parts)
            if s != f["amount_usd"]:
                errs.append(f"flow {f['id']}: amount {f['amount_usd']} != recomputed {s}")
    return errs


def backtest(d0, d1):
    a, b = build(d0, "backtest-from", False), build(d1, "backtest-to", False)
    def rows(l):
        return {r["id"]: r for k in ("flows", "observations", "events", "metrics") for r in l[k]}
    ra, rb = rows(a), rows(b)
    new = {k: v for k, v in rb.items() if k not in ra}
    changed = {k: (ra[k], rb[k]) for k in ra if k in rb and ra[k].get("amount_usd") != rb[k].get("amount_usd")}
    _, dec = load_raw()
    results = []
    for k in dec["known_events_since_backtest"]:
        m, hit, how = k["match"], None, None
        if m["kind"] == "observation":
            for r in new.values():
                if r.get("subject") == m["entity"] and r.get("metric") == m["metric"] and r.get("period_end") == m["period_end"]:
                    hit, how = r["id"], r["grade"]
        elif m["kind"] == "metric":
            for r in new.values():
                if r.get("entity") == m["entity"] and r.get("metric") == m["metric"] and r.get("as_of") == m["as_of"]:
                    hit, how = r["id"], r["grade"]
        elif m["kind"] == "event":
            if m["id"] in new:
                hit, how = m["id"], new[m["id"]]["grade"]
        elif m["kind"] == "flow_after":
            for r in new.values():
                if r.get("type") == m["type"] and r.get("payee") in m["payee_any"] and d(r["date"]) >= d(k["date"]) - dt.timedelta(days=7):
                    hit, how = r["id"], r["grade"]
        results.append({"id": k["id"], "date": k["date"], "what": k["what"], "independent_source": k["independent_source"],
                        "caught": bool(hit), "by_row": hit, "grade": how})
    return {"from": d0.isoformat(), "to": d1.isoformat(), "rows_at_from": len(ra), "rows_at_to": len(rb),
            "new_rows": sorted(new), "changed_amounts": {k: [x.get("amount_usd"), y.get("amount_usd")] for k, (x, y) in changed.items()},
            "known_events": results}


def headline(L):
    """The page's headline: equity and purchase commitments, as filed or company-stated (press-only rows are shown, not counted)."""
    return sum((f.get("amount_usd") or 0) for f in L["flows"] if f["type"] in ("equity", "purchase_commitment") and f.get("grade") != "reported")


def gate(live, new, pct=0.05, big=50e9):
    """Reasons to hold a run for review instead of publishing it: the headline moves more than pct, or one row moves by big or more."""
    h0, h1, why = headline(live), headline(new), []
    if h0 and abs(h1 - h0) / h0 > pct:
        why.append(f"headline moves {100 * (h1 - h0) / h0:+.1f}% (${h0 / 1e9:,.1f}B to ${h1 / 1e9:,.1f}B)")
    old = {f["id"]: f for f in live["flows"] if f["type"] not in NON_MONEY}
    now = {f["id"]: f for f in new["flows"] if f["type"] not in NON_MONEY}
    for i in sorted(set(old) | set(now)):
        a0 = (old.get(i) or {}).get("amount_usd") or 0
        a1 = (now.get(i) or {}).get("amount_usd") or 0
        if abs(a1 - a0) >= big:
            why.append(f"row {i} {'added' if i not in old else 'removed' if i not in now else 'changed'}: ${a0 / 1e9:,.1f}B to ${a1 / 1e9:,.1f}B")
    return why


# ====================== tokenization profile ======================
# Rows here are balances and ties, not deals: an amount outstanding at a date, who issues it, where it settles,
# what its reserves are held in. Same stages as the AI profile: raw -> schema -> validate -> gate -> ledger.json.
T_GRADES = {"onchain": "Read from the token contract through a public explorer; anyone can repeat it",
            "attested": "Accountant-signed reserve or examination report",
            "filed": "Regulator filing, register entry or official publication",
            "announced": "The issuer's or company's own page or announcement",
            "reported": "Press report; no primary source",
            "estimate": "Third-party estimate"}
T_COUNTED = ("onchain", "attested", "filed")           # grades the tokenized-money headline counts
T_COUNTED_ASSETS = T_COUNTED + ("announced",)          # tokenized assets also count a fund total its issuer states, marked as such
T_MONEY = ("equity", "debt", "issued", "settles", "holding")
T_TIES = ("issues", "tokenizes", "custody", "manages", "distributes", "partner", "licensed", "owns")
T_KINDS = ["company", "product", "ledger", "instrument", "investor", "government"]
T_OFFICIAL = ("fca.org.uk", "sec.gov", "govinfo.gov", "federalregister.gov", "occ.gov", "hkma.gov.hk", "europa.eu", "fdic.gov", "federalreserve.gov", "bis.org")
T_EXPLORERS = ("blockscout.com", "tronscanapi.com", "xrpscan.com")
T_ASSET_TYPES = {"tbills": "US Treasury bills and notes", "repo": "Treasury repo", "mmf": "Money market funds", "deposits": "Bank deposits and cash", "token": "Other tokens",
                 "gold": "Precious metals", "btc": "Bitcoin", "loans": "Secured loans", "govt": "Government securities, not split", "other": "Other"}
T_WRAPPERS = {"us_federal": "US national trust bank", "other_regulator": "Other regulator", "fund_registered": "Registered fund", "fund_private": "Private fund or note", "bank": "Bank deposit", "crypto_backed": "Crypto-collateralised, not counted", "none": "No charter on this ledger"}


def t_load():
    with open(os.path.join(RAW, "declared.json")) as f:
        dec = json.load(f)
    with open(os.path.join(RAW, "readings.csv")) as f:
        rd = list(csv.DictReader(f))
    for r in rd:
        r["value"] = int(r["value"])
    return dec, rd


def t_build(asof, run_id, with_history=True):
    dec, rd = t_load()
    E = {e["id"]: e for e in dec["entities"]}
    a = asof.isoformat()
    obs = []
    for r in rd:
        if r["available"] > a:
            continue
        inst, ch = r["instrument"], r["chain"]
        obs.append({
            "id": f"o-{inst}-{ch}-{r['as_of']}", "subject": inst, "metric": "outstanding" if ch == "all" else "supply_" + ch, "chain": ch,
            "period_start": None, "period_end": r["as_of"], "value_usd": r["value"], "basis": r["basis"], "method": r["method"],
            "derived": False, "derivation": None, "grade": r["grade"], "accn": r["method"], "form": None,
            "available": r["available"], "as_of": r["available"],
            "definition": (r["basis"][0].upper() + r["basis"][1:] + ". " + r["note"]).strip(), "source_url": r["source_url"], "source_title": r["source_title"],
        })
    obs.sort(key=lambda o: (o["subject"], o["chain"], o["period_end"], o["available"]))
    flows, events = [], []
    for f in dec["flows"]:
        if f["date"] > a:
            continue
        g = {k: v for k, v in f.items() if k != "revisions"}
        g["available"] = f["date"]; g.setdefault("as_of", f["date"])
        for r in sorted(f.get("revisions", []), key=lambda r: r["date"]):
            if r["date"] > a:
                break
            g.update({k: r[k] for k in ("amount_usd", "amount_qualifier", "grade", "source_title", "source_url", "status", "as_of") if k in r})
            g["available"] = r["date"]; g["definition"] = f["definition"] + " Update " + r["date"] + ": " + r["note"]
            events.append({"id": f"e-{f['id']}-r{r['date']}", "date": r["date"], "entities": [f["payer"], f["payee"]], "flow": f["id"], "level": f["level"],
                           "title": r["title"], "detail": r["note"], "grade": r.get("grade", f["grade"]), "source_title": r["source_title"], "source_url": r["source_url"]})
        flows.append(g)
        if f["type"] in T_MONEY or f["level"] == "map":
            events.append({"id": "e-" + f["id"], "date": f["date"], "entities": [f["payer"], f["payee"]], "flow": f["id"], "level": f["level"],
                           "title": f["title"], "detail": f["definition"], "grade": f["grade"], "source_title": f["source_title"], "source_url": f["source_url"]})
    # balances become lines: issuer -> token for the total, token -> chain for each chain read
    latest = {}
    for o in obs:
        k = (o["subject"], o["chain"]); p = latest.get(k)
        if not p or (o["period_end"], o["available"]) >= (p["period_end"], p["available"]):
            latest[k] = o
    for (inst, ch), o in sorted(latest.items()):
        e = E[inst]
        if ch == "all":
            flows.append({"id": f"f-{inst}-issued", "payer": e["parent"], "payee": inst, "type": "issued", "status": "cash_moved", "amount_usd": o["value_usd"],
                          "amount_qualifier": o["basis"], "date": o["available"], "available": o["available"], "as_of": o["period_end"], "grade": o["grade"],
                          "definition": o["definition"], "source_title": o["source_title"], "source_url": o["source_url"], "level": "map", "derived": True,
                          "derivation": "latest observation " + o["id"], "title": f"{e['name']} outstanding", "label": "Outstanding"})
        elif ch in E:
            flows.append({"id": f"f-{inst}-on-{ch}", "payer": inst, "payee": ch, "type": "settles", "status": "cash_moved", "amount_usd": o["value_usd"],
                          "amount_qualifier": o["basis"], "date": o["available"], "available": o["available"], "as_of": o["period_end"], "grade": o["grade"],
                          "definition": o["definition"], "source_title": o["source_title"], "source_url": o["source_url"], "level": "entity", "derived": True,
                          "derivation": "latest observation " + o["id"], "title": f"{e['name']} on {E[ch]['name']}", "label": "On " + E[ch]["name"]})
    have = {(f["payer"], f["payee"]) for f in flows if f["type"] == "settles"}
    for c in dec.get("chains_listed", []):
        if c["date"] <= a and (c["instrument"], c["chain"]) not in have:
            flows.append({"id": f"f-{c['instrument']}-on-{c['chain']}", "payer": c["instrument"], "payee": c["chain"], "type": "settles", "status": "cash_moved",
                          "amount_usd": None, "amount_qualifier": c.get("qualifier", "listed by the issuer; no amount per chain"), "date": c["date"], "available": c["date"], "as_of": c["date"],
                          "grade": c["grade"], "definition": c.get("definition", "The issuer lists this chain and publishes no separate amount for it."), "source_title": c["source_title"],
                          "source_url": c["source_url"], "level": "entity", "title": f"{E[c['instrument']]['name']} on {E[c['chain']]['name']}", "label": "On " + E[c["chain"]]["name"]})
    # every reading is an event on its token; a total that moves 10% or more against the last one is structural
    by = {}
    for o in obs:
        by.setdefault((o["subject"], o["chain"]), []).append(o)
    for (inst, ch), series in by.items():
        for i, o in enumerate(series):
            prev = series[i - 1] if i else None
            pct = (o["value_usd"] - prev["value_usd"]) / prev["value_usd"] * 100 if prev and prev["value_usd"] else None
            where = "" if ch == "all" else f" on {E[ch]['name'] if ch in E else ch}"
            events.append({"id": "e-" + o["id"], "date": o["available"], "entities": [inst], "flow": None, "observation": o["id"],
                           "level": "map" if ch == "all" and pct is not None and abs(pct) >= 10 else "entity",
                           "title": f"{E[inst]['name']}{where}: {t_fmt(o['value_usd'])} {o['basis']} at {o['period_end']}" + (f" ({pct:+.0f}% vs prior reading)" if pct is not None else ""),
                           "detail": o["definition"], "grade": o["grade"], "source_title": o["source_title"], "source_url": o["source_url"]})
    metrics = [dict(m) for m in dec.get("metrics", []) if m["available"] <= a]
    MN = {"reserve_assets": "reserve assets", "excess_reserves": "excess reserves", "usdc_on_platform": "USDC on platform", "distribution_costs_quarter": "distribution costs in the quarter"}
    for m in metrics:
        events.append({"id": "e-" + m["id"], "date": m["available"], "entities": [m["entity"]], "flow": None, "metric": m["id"], "level": "entity",
                       "title": f"{E[m['entity']]['name']} {MN.get(m['metric'], m['metric'])}: {t_fmt(m['value'])} as of {m['as_of']}", "detail": m["definition"],
                       "grade": m["grade"], "source_title": m["source_title"], "source_url": m["source_url"]})
    events += [dict(e) for e in dec["events"] if e["date"] <= a]
    events.sort(key=lambda e: (e["date"], e["id"]))
    # reconciliations
    checks, mm = [], {m["id"]: m for m in metrics}
    for (inst, ch), o in sorted(latest.items()):
        if ch != "all":
            continue
        parts = [x for x in obs if x["subject"] == inst and x["chain"] != "all" and x["period_end"] == o["period_end"] and x["grade"] == o["grade"]]
        if parts:
            s = sum(x["value_usd"] for x in parts)
            checks.append({"check": "chains_sum_to_total", "subject": inst, "metric": "outstanding", "period_end": o["period_end"], "computed": s, "stated": o["value_usd"], "abs_diff": abs(s - o["value_usd"]), "pass": abs(s - o["value_usd"]) <= len(parts)})   # parts converted from units and rounded to the dollar may differ by $1 each
    for c in dec.get("checks", []):
        tot = next((x for x in obs if x["subject"] == c["subject"] and x["chain"] == "all" and x["period_end"] == c["as_of"]), None)
        if not tot:
            continue
        if c["check"] == "reserves_cover_tokens" and c["reserves"] in mm:
            r = mm[c["reserves"]]["value"]
            checks.append({"check": c["check"], "subject": c["subject"], "metric": "outstanding", "period_end": c["as_of"], "computed": r, "stated": tot["value_usd"], "abs_diff": r - tot["value_usd"], "pass": r >= tot["value_usd"]})
        elif c["check"] == "attested_rounds_to_filed":
            got = round(tot["value_usd"] / c["round_to"]) * c["round_to"]
            checks.append({"check": c["check"], "subject": c["subject"], "metric": "outstanding", "period_end": c["as_of"], "computed": tot["value_usd"], "stated": c["stated"], "abs_diff": abs(got - c["stated"]), "pass": got == c["stated"], "basis": c["basis"], "source_url": c["source_url"]})
    # what backs each token: one row per reserve component, as the report prints it; the parts must sum to the reported reserves
    backing = [dict(b) for b in dec.get("backing", []) if b["available"] <= a]
    for b in backing:
        b["definition"] = f"{b['component']}: {t_fmt(abs(b['amount_usd']))}{' owed' if b['amount_usd'] < 0 else ''}, as printed in the report at {b['as_of']}. Held at: {b['held_label']}."
    for tok, asof_ in sorted({(b["token"], b["as_of"]) for b in backing}):
        part = [b for b in backing if b["token"] == tok and b["as_of"] == asof_]; m = mm.get(part[0]["reserve_metric"])
        if m:
            s = sum(b["amount_usd"] for b in part)
            checks.append({"check": "backing_sums_to_reserves", "subject": tok, "metric": "reserve_assets", "period_end": asof_, "computed": s, "stated": m["value"], "abs_diff": abs(s - m["value"]), "pass": abs(s - m["value"]) < 0.005})
    used = {x for f in flows for x in (f["payer"], f["payee"])} | {x for ev in events for x in ev["entities"]}
    for e in dec["entities"]:
        if e["id"] in used and e.get("parent"):
            used.add(e["parent"])
    ents = [e for e in dec["entities"] if e["id"] in used]
    ledger = {
        "schema_version": 1, "profile": "tokenization", "as_of": a, "window_start": dec["window_start"],
        "schema": {"grades": T_GRADES, "counted_grades": list(T_COUNTED), "counted_grades_assets": list(T_COUNTED_ASSETS), "flow_types": list(T_MONEY) + list(T_TIES), "statuses": STATUSES,
                   "levels": {"map": "Structural shift: changes who can issue, hold or settle what", "entity": "Any recorded change at one player"},
                   "asset_types": T_ASSET_TYPES, "wrappers": T_WRAPPERS,
                   "status_labels": {"cash_moved": "Live or closed", "contracted": "Signed, not closed", "announced": "Announced, applied for or conditional", "reported": "Press report"}},
        "entities": ents, "flows": flows, "observations": obs, "events": events, "ranges": [], "releases": [], "metrics": metrics,
        "coverage": [], "stress": dec.get("stress", {}), "checks": checks, "corrections": dec.get("corrections", []),
        "runs": [{"id": run_id, "as_of": a, "rows": len(flows) + len(obs) + len(events) + len(metrics) + len(backing)}],
        "watchlist": dec.get("watchlist", []), "gaps": dec.get("gaps", []), "pending_close": dec.get("pending_close", []), "backing": backing,
    }
    ledger["headline"] = t_headline(ledger)
    if with_history:
        dates = sorted({x["available"] for x in flows} | {o["available"] for o in obs} | {r["date"] for f in dec["flows"] for r in f.get("revisions", []) if r["date"] <= a})
        hist = {f["id"]: [] for f in flows}
        for ds in dates:
            snap = t_build(d(ds), run_id, with_history=False)
            for f in snap["flows"]:
                h = hist.get(f["id"])
                if h is None:
                    continue
                if not h or h[-1]["amount_usd"] != f["amount_usd"] or h[-1]["as_of"] != f["as_of"] or h[-1]["grade"] != f["grade"]:
                    h.append({"available": ds, "amount_usd": f["amount_usd"], "as_of": f["as_of"], "amount_qualifier": f["amount_qualifier"], "source_title": f["source_title"], "grade": f["grade"]})
        for f in flows:
            f["history"] = hist[f["id"]]
    return ledger


def t_fmt(v):
    return f"${v/1e9:,.2f}B" if v >= 1e9 else f"${v/1e6:,.1f}M"


def t_headline(L):
    """Two numbers, never summed. Tokenized money: fiat-backed dollar tokens and deposit tokens. Tokenized assets: funds, commodities and other claims
    on off-chain assets, less whatever counted tokens hold of each other. Money counts only balances read from chain, attested or filed; assets also
    count a fund total stated by its issuer, reported separately as assets_stated. Each token is counted once: its total where that total's grade
    counts, otherwise the sum of its counted chain reads (marked partial), otherwise its total as stated, shown and not counted. Tokens flagged
    counted=false (crypto-collateralised dollars) are shown and never counted."""
    E = {e["id"]: e for e in L["entities"]}
    cls_of = lambda i: "money" if E[i].get("asset_class") == "money" else "assets"
    ok_g = lambda c: T_COUNTED if c == "money" else T_COUNTED_ASSETS
    tot, ch = {}, {}
    for f in L["flows"]:
        if f["type"] == "issued" and E[f["payee"]]["kind"] == "product":
            tot[f["payee"]] = {"amount": f["amount_usd"], "grade": f["grade"], "as_of": f["as_of"], "row": f["id"], "partial": False}
        if f["type"] == "settles" and f["amount_usd"] is not None and f["grade"] in T_COUNTED and E[f["payer"]]["kind"] == "product":
            ch.setdefault(f["payer"], []).append(f)
    inst = {}
    for i in sorted(set(tot) | set(ch)):
        t, fs = tot.get(i), ch.get(i)
        if t and t["grade"] in ok_g(cls_of(i)):
            inst[i] = t
        elif fs:
            gs = {f["grade"] for f in fs}
            inst[i] = {"amount": sum(f["amount_usd"] for f in fs), "grade": gs.pop() if len(gs) == 1 else "onchain", "as_of": max(f["as_of"] for f in fs), "row": None,
                       "partial": True, "chains": sorted(f["payee"] for f in fs)}
            if t:
                inst[i].update({"stated_total": t["amount"], "stated_grade": t["grade"], "stated_as_of": t["as_of"]})
        else:
            inst[i] = t
    out = {"money": 0, "assets_gross": 0, "assets_stated": 0, "netting": 0, "stated_not_counted": {"money": 0, "assets": 0}, "shown_not_counted": {"money": 0, "assets": 0},
           "overlap_outside_count": 0, "by_instrument": {}, "netted_rows": [], "overlap_rows": []}
    dates = []
    for i, x in sorted(inst.items()):
        cls = cls_of(i)
        ok = bool(E[i].get("counted")) and x["grade"] in ok_g(cls)
        out["by_instrument"][i] = {**x, "class": cls, "counted": ok}
        if ok:
            out["money" if cls == "money" else "assets_gross"] += x["amount"]; dates.append(x["as_of"])
            if cls == "assets" and x["grade"] == "announced": out["assets_stated"] += x["amount"]
        elif not E[i].get("counted"):
            out["shown_not_counted"][cls] += x["amount"]
        elif x["grade"] == "announced":
            out["stated_not_counted"][cls] += x["amount"]
    for f in L["flows"]:
        if f["type"] != "holding" or f["amount_usd"] is None:
            continue
        a, b = out["by_instrument"].get(f["payer"]), out["by_instrument"].get(f["payee"])
        if not (a and b and b["class"] == "assets"):
            continue   # reserves held outside this ledger's tokens: nothing is counted twice
        if a["counted"] and b["counted"] and not b["partial"]:
            n = min(f["amount_usd"], b["amount"]); out["netting"] += n; out["netted_rows"].append(f["id"])
        else:
            out["overlap_outside_count"] += f["amount_usd"]; out["overlap_rows"].append(f["id"])
    out["assets"] = out["assets_gross"] - out["netting"]
    out["as_of_range"] = [min(dates), max(dates)] if dates else None
    return out


def t_validate(L):
    errs, ids = [], set()
    ent = {e["id"]: e for e in L["entities"]}

    def need(row, kind, fields):
        for f in fields:
            if row.get(f) in (None, ""):
                errs.append(f"{kind} {row.get('id')}: missing {f}")
        g, u = row.get("grade"), row.get("source_url") or ""
        if g not in T_GRADES:
            errs.append(f"{kind} {row.get('id')}: bad grade {g}")
        if not re.match(r"^https://[^\s]+\.[^\s]+$", u):
            errs.append(f"{kind} {row.get('id')}: bad source_url")
        if g == "filed" and not any(h in u for h in T_OFFICIAL):
            errs.append(f"{kind} {row.get('id')}: graded filed without an official source")
        if g == "onchain" and not any(h in u for h in T_EXPLORERS):
            errs.append(f"{kind} {row.get('id')}: graded onchain without an explorer source")
        if row.get("id") in ids:
            errs.append(f"duplicate id {row.get('id')}")
        ids.add(row.get("id"))

    for e in L["entities"]:
        if e.get("kind") not in T_KINDS: errs.append(f"entity {e['id']}: bad kind {e.get('kind')}")
        if e.get("parent") and e["parent"] not in ent: errs.append(f"entity {e['id']}: unknown parent")
        if e.get("kind") == "product" and (e.get("asset_class") is None or e.get("counted") is None): errs.append(f"entity {e['id']}: a token needs asset_class and counted")
    for f in L["flows"]:
        need(f, "flow", ["id", "payer", "payee", "type", "status", "date", "as_of", "definition", "source_url", "source_title"])
        if f.get("type") not in T_MONEY + T_TIES: errs.append(f"flow {f['id']}: bad type")
        if f.get("type") in T_TIES and f.get("amount_usd") is not None: errs.append(f"flow {f['id']}: a {f['type']} tie cannot carry an amount")
        if f.get("status") not in STATUSES: errs.append(f"flow {f['id']}: bad status")
        if f.get("payer") not in ent or f.get("payee") not in ent: errs.append(f"flow {f['id']}: unknown entity")
        if f.get("type") in T_MONEY and f.get("amount_usd") is None and not f.get("amount_qualifier"): errs.append(f"flow {f['id']}: no amount and no qualifier")
        if f.get("as_of") and f["as_of"] > f["available"]: errs.append(f"flow {f['id']}: describes a date after it was published")
    omap = {o["id"]: o for o in L["observations"]}
    for o in L["observations"]:
        need(o, "observation", ["id", "subject", "metric", "chain", "period_end", "value_usd", "basis", "method", "available", "definition", "source_url"])
        if o["subject"] not in ent: errs.append(f"observation {o['id']}: unknown token")
        if o["period_end"] > o["available"]: errs.append(f"observation {o['id']}: describes a date after it was published")
    for e in L["events"]:
        need(e, "event", ["id", "date", "entities", "level", "title", "source_url"])
        for x in e.get("entities", []):
            if x not in ent: errs.append(f"event {e['id']}: unknown entity {x}")
    for m in L.get("metrics", []):
        need(m, "metric", ["id", "entity", "metric", "value", "unit", "as_of", "available", "definition", "source_url", "source_title"])
        if m["entity"] not in ent: errs.append(f"metric {m['id']}: unknown entity")
    for b in L.get("backing", []):
        need(b, "backing", ["id", "token", "asset_type", "component", "amount_usd", "held_label", "as_of", "available", "definition", "source_url", "source_title"])
        if b["token"] not in ent: errs.append(f"backing {b['id']}: unknown token")
        for k in ("held_at", "counterpart"):
            if b.get(k) and b[k] not in ent: errs.append(f"backing {b['id']}: unknown {k}")
        if b["asset_type"] not in T_ASSET_TYPES: errs.append(f"backing {b['id']}: bad asset_type")
    for e in L["entities"]:
        if e.get("kind") == "product" and (e.get("wrapper") or {}).get("regime") not in T_WRAPPERS: errs.append(f"entity {e['id']}: a token needs a wrapper")
    for c in L["checks"]:
        if not c["pass"]:
            errs.append(f"reconciliation failed: {c}")
    for f in L["flows"]:
        if f.get("derivation", "").startswith("latest observation "):
            o = omap.get(f["derivation"][len("latest observation "):])
            if not o or o["value_usd"] != f["amount_usd"]:
                errs.append(f"flow {f['id']}: amount does not match the observation it cites")
    if L.get("headline") != t_headline(L):
        errs.append("headline does not recompute from the rows")
    return errs


def t_gate(live, new, pct=0.05, big=5e9, rowpct=0.20, newbar=1e9):
    """Hold a run for review when either headline moves more than pct, one token moves by big or by rowpct,
    a token enters above newbar, or a netting row is added, removed or changed. Failed reconciliations never reach here: build refuses them."""
    why, h0, h1 = [], live["headline"], new["headline"]
    for k, name in (("money", "tokenized money"), ("assets", "tokenized assets")):
        if h0[k] and abs(h1[k] - h0[k]) / h0[k] > pct:
            why.append(f"{name} moves {100 * (h1[k] - h0[k]) / h0[k]:+.1f}% ({t_fmt(h0[k])} to {t_fmt(h1[k])})")
    a, b = h0["by_instrument"], h1["by_instrument"]
    for i in sorted(set(a) | set(b)):
        x, y = (a.get(i) or {}).get("amount") or 0, (b.get(i) or {}).get("amount") or 0
        if i not in a and y >= newbar:
            why.append(f"{i} enters at {t_fmt(y)}")
        elif i in a and (abs(y - x) >= big or (x and abs(y - x) / x >= rowpct)):
            why.append(f"{i} moves {t_fmt(x)} to {t_fmt(y)}")
    hold = lambda L: {f["id"]: f["amount_usd"] for f in L["flows"] if f["type"] == "holding"}
    p, q = hold(live), hold(new)
    for i in sorted(set(p) | set(q)):
        if p.get(i) != q.get(i) and (i in h0["netted_rows"] or i in h1["netted_rows"]):
            why.append(f"netting row {i} {'added' if i not in p else 'removed' if i not in q else 'changed'}")
    return why


def profile_of():
    with open(os.path.join(RAW, "declared.json")) as f:
        return json.load(f).get("profile", "ai")


if __name__ == "__main__":
    ap = argparse.ArgumentParser(description="One harness, a profile per ledger. The profile is read from raw/declared.json; a ledger file names its own.")
    ap.add_argument("--raw", help="folder holding declared.json and the raw readings (default: ../raw beside this file)")
    sub = ap.add_subparsers(dest="cmd", required=True)
    b = sub.add_parser("build"); b.add_argument("--asof", required=True); b.add_argument("--out", required=True); b.add_argument("--run-id", default="manual")
    v = sub.add_parser("validate"); v.add_argument("ledger")
    gt = sub.add_parser("gate"); gt.add_argument("--live", required=True); gt.add_argument("--new", required=True); gt.add_argument("--pct", type=float, default=0.05); gt.add_argument("--big", type=float)
    t = sub.add_parser("backtest"); t.add_argument("--from", dest="d0", required=True); t.add_argument("--to", dest="d1", required=True); t.add_argument("--out")
    a = ap.parse_args()
    if a.raw:
        RAW = os.path.abspath(a.raw)
    if a.cmd == "build":
        tok = profile_of() == "tokenization"
        L = t_build(d(a.asof), a.run_id) if tok else build(d(a.asof), a.run_id)
        errs = t_validate(L) if tok else validate(L)
        if errs:
            print("VALIDATION FAILED"); [print(" -", e) for e in errs]; sys.exit(1)
        if not tok:
            L["backtest"] = backtest(d("2026-09-01"), d(a.asof))
        vf = os.path.join(os.path.dirname(os.path.abspath(a.out)), "verification.json")
        L["verification"] = json.load(open(vf)) if os.path.exists(vf) else None
        if L["verification"]:
            dec = (t_load()[0] if tok else load_raw()[1])
            L["verification"]["addressed"] = dec.get("verification_addressed", [])
            L["verification"]["pending"] = dec.get("verification_pending", [])
        json.dump(L, open(a.out, "w"), indent=1)
        print(f"wrote {a.out}: {len(L['entities'])} entities, {len(L['flows'])} flows, {len(L['observations'])} observations, {len(L['events'])} events, {len(L['checks'])} reconciliations passed")
        if tok:
            h = L["headline"]; print(f"headline: money {t_fmt(h['money'])}; assets {t_fmt(h['assets'])} (gross {t_fmt(h['assets_gross'])}, of which issuer-stated {t_fmt(h['assets_stated'])}; netted {t_fmt(h['netting'])}); shown, not counted: money {t_fmt(h['shown_not_counted']['money'])}")
    elif a.cmd == "validate":
        L = json.load(open(a.ledger)); errs = t_validate(L) if L.get("profile") == "tokenization" else validate(L)
        print("OK" if not errs else "FAILED"); [print(" -", e) for e in errs]; sys.exit(1 if errs else 0)
    elif a.cmd == "gate":
        live, new = json.load(open(a.live)), json.load(open(a.new))
        why = t_gate(live, new, a.pct, a.big or 5e9) if new.get("profile") == "tokenization" else gate(live, new, a.pct, a.big or 50e9)
        print("HOLD" if why else "PASS"); [print(" -", w) for w in why]; sys.exit(3 if why else 0)
    else:
        r = backtest(d(a.d0), d(a.d1))
        if a.out: json.dump(r, open(a.out, "w"), indent=1)
        print(json.dumps(r, indent=1))
